            /*
                Address of the global `hltv` (CHLTVServer*) variable in engine.

                The plugin dereferences this via LoadAddressFromAddress to get the
                current CHLTVServer instance (NULL when SourceTV is disabled).

                On 32-bit builds the engine encodes `&hltv` as a plain 4-byte imm32
                baked into a `mov reg, &hltv` instruction, so SourceMod's
                sig+`read N` mechanism can hand us `&hltv` directly. On 64-bit
                builds the encoding is RIP-relative (`mov reg, [rip+disp32]` /
                `lea reg, [rip+disp32]`) so `read N` would dereference the wrong
                8 bytes — the shim instead consumes "CHLTVServer*_x64_disp32"
                below and computes &hltv = (disp32_ptr + 4) + disp32 in SP.

                ---- linux x86 ----
                anchor: CHLTVDemoRecorder::StartAutoRecording @ 0x12F660 has

                    .text:0012F67D     mov     eax, ds:hltv     ; A1 1C BD 33 00

                The `A1` opcode is one byte at func+0x1D, so the imm32 (= &hltv) sits
                at func+0x1E (= 30 decimal). `read 30` makes SM read those 4 bytes and
                hand us &hltv directly.

                ---- windows x86 ----
                anchor: tv_stoprecord_f (ConCommand callback) @ 0x1017F7E0 starts with

                    .text:1017F7E0     mov     ecx, dword_10635664   ; 8B 0D 64 56 63 10

                The `8B 0D` opcode is two bytes, so the imm32 (= &hltv) sits at func+2.
                `read 2` hands us &hltv directly. See the "tv_stoprecord_f" entry in
                Signatures below for the full disasm.
            */
            "CHLTVServer*"
            {
                "linux"
                {
                    "signature" "CHLTVDemoRecorder::StartAutoRecording"
                    "read"      "30"
                }
                "windows"
                {
                    "signature" "tv_stoprecord_f"
                    "read"      "2"
                }
                // linux64 / windows64 use the RIP-relative path -- see
                // "CHLTVServer*_x64_disp32" below.
            }

            /*
                x64 fallback for &hltv -- returns the address of the 4-byte
                disp32 *inside* the RIP-relative LEA, NOT &hltv itself. The
                shim reads those 4 bytes as a signed int32 and resolves

                    &hltv = (disp32_ptr + 4) + disp32

                (the `+ 4` is the size of the disp32, since RIP after the
                instruction executes is the byte just past the disp32.)

                We use SourceMod's `offset` keyword (vs `read`) so SM does
                sigMatch + N WITHOUT dereferencing -- on x64 that "would"
                pull in 8 bytes of (disp32 || next-instruction-bytes), which
                is meaningless.

                Per-platform disasm + offset derivation lives next to the
                relevant sub-block below.
            */
            "CHLTVServer*_x64_disp32"
            {
                /*
                    ---- linux x64 ----
                    anchor: CHLTVDemoRecorder::StartAutoRecording @ 0x10B320 has

                        .text:0010B349     lea     rax, hltv     ; 48 8D 05 20 E3 25 00

                    The `48 8D 05` REX+opcode+ModRM is three bytes at func+0x29,
                    so the disp32 sits at func+0x2C (= 44 decimal). The next
                    instruction starts at func+0x30, which is also (disp32_ptr + 4).
                    Verified at runtime: 0x10B350 + 0x25E320 = 0x369670 == &hltv.
                */
                "linux64"
                {
                    "signature" "CHLTVDemoRecorder::StartAutoRecording"
                    "offset"    "44"
                }

                /*
                    ---- windows x64 ----
                    anchor: tv_stoprecord_f (engine.dll x64 sub_18019EC00,
                    0x4D-byte ConCommand callback). Function prologue:

                        .text:18019EC00   48 83 EC 28        sub  rsp, 28h
                        .text:18019EC04   48 8B 0D <disp32>  mov  rcx, &hltv
                        .text:18019EC0B   48 85 C9           test rcx, rcx

                    The 4-byte `sub rsp, 28h` prologue + the 3-byte
                    `48 8B 0D` REX+opcode+ModRM puts the disp32 at func+7.
                    Verified live: 0x18019EC0B + 0x005B3F15 = 0x180752B20
                    (= &hltv = qword_180752B20).
                */
                "windows64"
                {
                    "signature" "tv_stoprecord_f"
                    "offset"    "7"
                }
            }
